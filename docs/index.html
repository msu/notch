Notch

A scripting language for Java & Minecraft, from Montana State University

example.notch
import java.time.LocalDate

today = LocalDate.now()

for name in ['Alice', 'Bob', 'Carol']
  print('Hello, ' + name + '! Today is ' + today)
end
